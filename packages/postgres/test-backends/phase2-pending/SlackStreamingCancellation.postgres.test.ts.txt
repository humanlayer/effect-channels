import { assert, describe, it } from '@effect/vitest'
import { bind, MailboxStore } from '@humanlayer/channels-delivery'
import { ConfigProvider, Deferred, Effect, Fiber, Layer, Queue, Schema } from 'effect'

import { defaultDeliveryPolicy, StorageTypeId } from '../src/index.ts'
import { PostgresStorage } from '../src/postgres.ts'
import { PostgresTestDatabase, postgresEmulatorTestsEnabled } from './support/PostgresTestDatabase.ts'

const Event = Schema.Struct({ id: Schema.String })

describe.skipIf(!postgresEmulatorTestsEnabled)('Slack stream cancellation with Postgres ownership', () => {
	it.live(
		'a non-owner cancels an active attempt; duplicate Stop does not cancel the next',
		() =>
			Effect.gen(function* () {
				const database = yield* PostgresTestDatabase
				const storage = () =>
					PostgresStorage.internal()[StorageTypeId].pipe(
						Layer.provide(ConfigProvider.layer(ConfigProvider.fromUnknown({ DATABASE_URL: database.url }))),
						Layer.fresh,
					)
				const calls = yield* Queue.unbounded<string>()
				const finalized = yield* Queue.unbounded<string>()
				const release = yield* Deferred.make<void>()
				const delivery = bind({
					namespace: 'app-cancellation',
					handlerId: 'stream',
					policy: { ...defaultDeliveryPolicy, heartbeatMs: 25 },
					definition: {
						name: 'stream',
						version: '1',
						provider: 'slack',
						event: Event,
						resource: Schema.String,
						resourceKey: (resource) => resource,
						identify: (event) => ({ eventId: event.id, installation: 'T_TEST', resource: 'thread' }),
					},
					handler: (event) =>
						Queue.offer(calls, event.id).pipe(
							Effect.andThen(Deferred.await(release)),
							Effect.ensuring(Queue.offer(finalized, event.id)),
						),
				})
				yield* Effect.gen(function* () {
					const receipt = yield* delivery.admit({ event: { id: 'A' } })
					const first = yield* delivery.processMailbox(receipt).pipe(Effect.forkChild)
					assert.strictEqual(yield* Queue.take(calls), 'A')
					yield* delivery.admit({ event: { id: 'B' } })
					assert.strictEqual(
						yield* delivery.cancelActive({ ...receipt, controlId: 'stop' }).pipe(Effect.provide(storage())),
						true,
					)
					yield* Fiber.join(first)
					assert.strictEqual(yield* Queue.take(finalized), 'A')
					const store = yield* MailboxStore
					assert.ok(
						(yield* store.loadMailbox(receipt))?.state.outcomes.some(
							(outcome) => outcome.kind === 'cancelled',
						),
					)
					const second = yield* delivery.processMailbox(receipt).pipe(Effect.forkChild)
					assert.strictEqual(yield* Queue.take(calls), 'B')
					assert.strictEqual(
						yield* delivery.cancelActive({ ...receipt, controlId: 'stop' }).pipe(Effect.provide(storage())),
						false,
					)
					yield* Deferred.succeed(release, undefined)
					yield* Fiber.join(second)
					assert.strictEqual(yield* Queue.take(finalized), 'B')
				}).pipe(Effect.provide(storage()))
			}).pipe(Effect.provide(PostgresTestDatabase.layer)),
		30_000,
	)
})
