import { assert, describe, layer } from '@effect/vitest'
import { MarkdownContent } from '@humanlayer/channels-slack'
import { SlackConnection, SlackConnectionCredentials } from '@humanlayer/channels-slack'
import { ConfigProvider, Deferred, Effect, Layer, Queue, Redacted, Ref } from 'effect'

import { defaultDeliveryPolicy, createChannelsApp, slack } from '../src/index.ts'
import { PostgresStorage } from '../src/postgres.ts'
import { PostgresTestDatabase, postgresEmulatorTestsEnabled } from './support/PostgresTestDatabase.ts'
import {
	HistoryResponse,
	PostedMessageResponse,
	SlackEmulator,
	slackEmulatorAliceToken,
	slackEmulatorBotId,
	slackEmulatorBotToken,
	slackEmulatorBotUserId,
	slackEmulatorSigningSecret,
} from './support/SlackEmulator.ts'

interface SlackMessageEventInput {
	readonly teamId: string
	readonly eventId: string
	readonly channelId: string
	readonly ts: string
	readonly text: string
	readonly userId: string
	readonly threadTs?: string
}

interface SlackMessageEventPayload {
	type: 'app_mention' | 'message'
	channel: string
	ts: string
	text: string
	user: string
	thread_ts?: string
}

const eventCallback = (input: SlackMessageEventInput) => {
	const event: SlackMessageEventPayload = {
		type: input.threadTs === undefined ? 'app_mention' : 'message',
		channel: input.channelId,
		ts: input.ts,
		text: input.text,
		user: input.userId,
	}
	if (input.threadTs !== undefined) {
		event.thread_ts = input.threadTs
	}
	return {
		type: 'event_callback' as const,
		team_id: input.teamId,
		event_id: input.eventId,
		event_time: 1,
		event,
	}
}

describe.skipIf(!postgresEmulatorTestsEnabled)('Slack emulator shared Postgres integration', () => {
	layer(Layer.merge(SlackEmulator.layer, PostgresTestDatabase.layer), { timeout: '30 seconds' })((it) => {
		it.effect(
			'persists subscription and FIFO ownership across two production application runtimes',
			() =>
				Effect.gen(function* () {
					const emulator = yield* SlackEmulator
					const database = yield* PostgresTestDatabase
					const deliveries = yield* Queue.unbounded<string>()
					const mentionCount = yield* Ref.make(0)
					const followUpCount = yield* Ref.make(0)
					const activeHandlers = yield* Ref.make(0)
					const maxActiveHandlers = yield* Ref.make(0)
					const runtimeAEntered = yield* Deferred.make<void>()
					const releaseRuntimeA = yield* Deferred.make<void>()
					const followUpEntered = yield* Deferred.make<void>()
					const connection = SlackConnection.make({
						credentials: SlackConnectionCredentials.make({
							botToken: Redacted.make(slackEmulatorBotToken),
							botUserId: slackEmulatorBotUserId,
							botId: slackEmulatorBotId,
						}),
					})
					const enterHandler = Ref.updateAndGet(activeHandlers, (count) => count + 1).pipe(
						Effect.tap((active) => Ref.update(maxActiveHandlers, (maximum) => Math.max(maximum, active))),
					)
					const leaveHandler = Ref.update(activeHandlers, (count) => count - 1)
					const makeApp = (runtime: 'A' | 'B') =>
						createChannelsApp({
							providers: [slack({ loadConnection: () => Effect.succeed(connection) })],
							storage: PostgresStorage.internal(),
							policy: {
								...defaultDeliveryPolicy,
								mode: 'serial',
								leaseMs: 1_000,
								heartbeatMs: 250,
								retryBaseMs: 50,
								retryMaxMs: 200,
							},
							handlers: {
								onNewMention: (thread, message) =>
									Effect.gen(function* () {
										yield* enterHandler
										yield* Ref.update(mentionCount, (count) => count + 1)
										yield* thread.subscribe()
										if (runtime === 'A') {
											yield* Deferred.succeed(runtimeAEntered, undefined)
											yield* Deferred.await(releaseRuntimeA)
										}
										yield* thread.post(
											MarkdownContent.make({ markdown: `runtime ${runtime} mention` }),
										)
										yield* Queue.offer(deliveries, `${runtime}:mention:${message.ref}`)
										yield* leaveHandler
									}),
								onSubscribedMessage: (thread, message) =>
									Effect.gen(function* () {
										yield* enterHandler
										yield* Ref.update(followUpCount, (count) => count + 1)
										yield* Deferred.succeed(followUpEntered, undefined)
										yield* thread.post(
											MarkdownContent.make({ markdown: `runtime ${runtime} follow-up` }),
										)
										yield* Queue.offer(deliveries, `${runtime}:follow-up:${message.ref}`)
										yield* leaveHandler
									}),
							},
							advanced: {
								slackApiOrigin: new URL(`${emulator.emulator.url}/api`),
								configProvider: ConfigProvider.fromUnknown({
									DATABASE_URL: database.url,
									SLACK_SIGNING_SECRET: slackEmulatorSigningSecret,
									SLACK_BOT_USER_ID: slackEmulatorBotUserId,
									SLACK_BOT_ID: slackEmulatorBotId,
								}),
							},
						})
					const appA = makeApp('A')
					yield* Effect.addFinalizer(() => Effect.promise(() => appA.close()))
					const root = yield* emulator.call(
						slackEmulatorAliceToken,
						'chat.postMessage',
						{ channel: emulator.publicChannelId, text: `<@${slackEmulatorBotUserId}> durable hello` },
						PostedMessageResponse,
					)
					const mentionResponse = yield* Effect.promise(() =>
						appA.handle(
							emulator.signedWebhook(
								eventCallback({
									teamId: emulator.teamId,
									eventId: 'Ev_postgres_runtime_a',
									channelId: root.channel,
									ts: root.ts,
									text: root.message.text,
									userId: root.message.user ?? emulator.aliceUserId,
								}),
							),
						),
					)
					assert.strictEqual(mentionResponse.status, 200)
					yield* Deferred.await(runtimeAEntered)
					assert.strictEqual(yield* Ref.get(activeHandlers), 1)
					assert.strictEqual(yield* Queue.size(deliveries), 0)

					const appB = makeApp('B')
					yield* Effect.addFinalizer(() => Effect.promise(() => appB.close()))
					const followUp = yield* emulator.call(
						slackEmulatorAliceToken,
						'chat.postMessage',
						{ channel: emulator.publicChannelId, thread_ts: root.ts, text: 'durable follow-up' },
						PostedMessageResponse,
					)
					const followUpResponse = yield* Effect.promise(() =>
						appB.handle(
							emulator.signedWebhook(
								eventCallback({
									teamId: emulator.teamId,
									eventId: 'Ev_postgres_runtime_b',
									channelId: followUp.channel,
									ts: followUp.ts,
									threadTs: root.ts,
									text: followUp.message.text,
									userId: followUp.message.user ?? emulator.aliceUserId,
								}),
							),
						),
					)
					assert.strictEqual(followUpResponse.status, 200)
					assert.strictEqual(yield* Deferred.isDone(followUpEntered), false)
					assert.strictEqual(yield* Ref.get(mentionCount), 1)
					assert.strictEqual(yield* Ref.get(followUpCount), 0)
					assert.strictEqual(yield* Ref.get(activeHandlers), 1)
					yield* Deferred.succeed(releaseRuntimeA, undefined)
					assert.strictEqual(yield* Queue.take(deliveries), `A:mention:${root.ts}`)
					const followUpDelivery = yield* Queue.take(deliveries)
					assert.strictEqual(followUpDelivery.endsWith(`:follow-up:${followUp.ts}`), true)
					const replies = yield* emulator.call(
						slackEmulatorBotToken,
						'conversations.replies',
						{ channel: emulator.publicChannelId, ts: root.ts },
						HistoryResponse,
					)
					const botReplies = replies.messages
						.filter((message) => message.user === slackEmulatorBotUserId)
						.map((message) => message.text)
					assert.deepStrictEqual(botReplies.slice(0, 1), ['runtime A mention'])
					assert.strictEqual(botReplies.at(1)?.endsWith(' follow-up'), true)
					assert.strictEqual(yield* Ref.get(mentionCount), 1)
					assert.strictEqual(yield* Ref.get(followUpCount), 1)
					assert.strictEqual(yield* Ref.get(maxActiveHandlers), 1)
					assert.strictEqual(yield* Ref.get(activeHandlers), 0)
					assert.strictEqual(yield* Queue.size(deliveries), 0)
					yield* Effect.promise(() => appA.close())
					yield* Effect.promise(() => appB.close())
				}),
			{ timeout: 30_000 },
		)
	})
})
