import { assert, it } from '@effect/vitest'
import { Effect, Layer, Queue } from 'effect'

import {
	Channels,
	ChannelsGate,
	ChannelsObserver,
	DirectMessageOpenFailed,
	EphemeralFallbackToDm,
	EphemeralNoFallback,
	EphemeralResult,
	MarkdownContent,
	ProviderRegistry,
	SentMessage,
	SentRef,
	Thread,
	type OutboundReport,
} from '../src/index'
import {
	makeCoreDependencies,
	makeFakeProvider,
	noCapabilities,
	expectTaggedFailure,
	testAuthor,
	testMessage,
	testThreadRef,
} from './support'

const dmThread = Thread.fromRef({
	...testThreadRef,
	channel: { ...testThreadRef.channel, isDm: true },
})

const sent = SentMessage.make({
	ref: SentRef.make({
		threadId: dmThread.ref.id,
		messageRef: testMessage.ref,
		provider: 'slack',
		degraded: [],
	}),
	message: testMessage,
})

const makeObservedLayer = (
	reports: Queue.Queue<OutboundReport>,
	gate: Layer.Layer<ChannelsGate> = ChannelsGate.layerAllowAll,
) => {
	const observer = ChannelsObserver.make({
		eventReceived: () => Effect.void,
		outboundSent: (report) => Queue.offer(reports, report).pipe(Effect.asVoid),
	})
	const dependencies = makeCoreDependencies(observer, gate)
	return Layer.merge(Channels.layer(), dependencies).pipe(Layer.provideMerge(dependencies))
}

it.effect('reports openDM success with its canonical result identity', () =>
	Effect.gen(function* () {
		const reports = yield* Queue.unbounded<OutboundReport>()
		yield* Effect.gen(function* () {
			const registry = yield* ProviderRegistry
			yield* registry.register(
				makeFakeProvider({
					capabilities: { ...noCapabilities, directMessages: { ingress: true, open: true } },
					openDM: () => Effect.succeed(dmThread),
				}),
			)
			const channels = yield* Channels
			yield* channels.openDM({ provider: 'slack', tenant: testThreadRef.channel.tenant, user: testAuthor })
		}).pipe(Effect.provide(makeObservedLayer(reports)))
		const report = yield* Queue.take(reports)
		assert.strictEqual(report.operation, 'open_dm')
		assert.strictEqual(report.ok, true)
		assert.strictEqual(report.threadId, dmThread.ref.id)
	}),
)

it.effect('preserves non-retryable tenant denial for openDM and postEphemeral', () =>
	Effect.gen(function* () {
		const reports = yield* Queue.unbounded<OutboundReport>()
		const denied = ChannelsGate.make(() => Effect.succeed(false))
		yield* Effect.gen(function* () {
			const registry = yield* ProviderRegistry
			yield* registry.register(
				makeFakeProvider({
					capabilities: {
						...noCapabilities,
						directMessages: { ingress: true, open: true },
						ephemeral: { native: true, dmFallback: true },
					},
				}),
			)
			const channels = yield* Channels
			const openError = yield* expectTaggedFailure('DirectMessageOpenFailed')(
				channels.openDM({ provider: 'slack', tenant: testThreadRef.channel.tenant, user: testAuthor }),
			)
			assert.strictEqual(openError.retryability, 'non_retryable')
			const ephemeralError = yield* expectTaggedFailure('PostFailed')(
				channels.postEphemeral({
					threadId: testThreadRef.id,
					user: testAuthor,
					content: MarkdownContent.make({ markdown: 'private' }),
					fallback: EphemeralNoFallback.make({}),
				}),
			)
			assert.strictEqual(ephemeralError.retryability, 'non_retryable')
		}).pipe(Effect.provide(makeObservedLayer(reports, denied)))
	}),
)

it.effect('preserves non-retryable tenant denial from the persistent DM fallback post', () =>
	Effect.gen(function* () {
		const reports = yield* Queue.unbounded<OutboundReport>()
		const decisions = yield* Queue.unbounded<boolean>()
		yield* Queue.offerAll(decisions, [true, true, false])
		const gate = ChannelsGate.make(() => Queue.take(decisions))
		yield* Effect.gen(function* () {
			const registry = yield* ProviderRegistry
			yield* registry.register(
				makeFakeProvider({
					capabilities: {
						...noCapabilities,
						directMessages: { ingress: true, open: true },
						ephemeral: { native: false, dmFallback: true },
					},
					openDM: () => Effect.succeed(dmThread),
				}),
			)
			const channels = yield* Channels
			const error = yield* expectTaggedFailure('PostFailed')(
				channels.postEphemeral({
					threadId: testThreadRef.id,
					user: testAuthor,
					content: MarkdownContent.make({ markdown: 'fallback' }),
					fallback: EphemeralFallbackToDm.make({}),
				}),
			)
			assert.strictEqual(error.retryability, 'non_retryable')
		}).pipe(Effect.provide(makeObservedLayer(reports, gate)))
	}),
)

it.effect('reports openDM failure without inventing a thread identity', () =>
	Effect.gen(function* () {
		const reports = yield* Queue.unbounded<OutboundReport>()
		yield* Effect.gen(function* () {
			const registry = yield* ProviderRegistry
			yield* registry.register(
				makeFakeProvider({
					capabilities: { ...noCapabilities, directMessages: { ingress: true, open: true } },
					openDM: (input) =>
						Effect.fail(
							DirectMessageOpenFailed.make({
								provider: input.provider,
								tenant: input.tenant,
								userId: input.user.userId,
								message: 'unavailable',
							}),
						),
				}),
			)
			const channels = yield* Channels
			yield* expectTaggedFailure('DirectMessageOpenFailed')(
				channels.openDM({ provider: 'slack', tenant: testThreadRef.channel.tenant, user: testAuthor }),
			)
		}).pipe(Effect.provide(makeObservedLayer(reports)))
		const report = yield* Queue.take(reports)
		assert.strictEqual(report.operation, 'open_dm')
		assert.strictEqual(report.ok, false)
		assert.strictEqual(report.threadId, undefined)
	}),
)

it.effect('reports native and fallback ephemeral outcomes', () =>
	Effect.gen(function* () {
		const reports = yield* Queue.unbounded<OutboundReport>()
		const nativeProvider = makeFakeProvider({
			capabilities: { ...noCapabilities, ephemeral: { native: true, dmFallback: false } },
			postEphemeral: () => Effect.succeed(EphemeralResult.make({ sent, usedFallback: false })),
		})
		yield* Effect.gen(function* () {
			const registry = yield* ProviderRegistry
			yield* registry.register(nativeProvider)
			const channels = yield* Channels
			yield* channels.postEphemeral({
				threadId: testThreadRef.id,
				user: testAuthor,
				content: MarkdownContent.make({ markdown: 'native' }),
				fallback: EphemeralNoFallback.make({}),
			})
		}).pipe(Effect.provide(makeObservedLayer(reports)))
		const native = yield* Queue.take(reports)
		assert.strictEqual(native.operation, 'post_ephemeral')
		assert.strictEqual(native.outcome, 'native')
		assert.strictEqual(native.ok, true)

		const fallbackReports = yield* Queue.unbounded<OutboundReport>()
		yield* Effect.gen(function* () {
			const registry = yield* ProviderRegistry
			yield* registry.register(
				makeFakeProvider({
					capabilities: {
						...noCapabilities,
						directMessages: { ingress: true, open: true },
						ephemeral: { native: false, dmFallback: true },
					},
					openDM: () => Effect.succeed(dmThread),
					post: () => Effect.succeed(sent),
				}),
			)
			const channels = yield* Channels
			yield* channels.postEphemeral({
				threadId: testThreadRef.id,
				user: testAuthor,
				content: MarkdownContent.make({ markdown: 'fallback' }),
				fallback: EphemeralFallbackToDm.make({}),
			})
		}).pipe(Effect.provide(makeObservedLayer(fallbackReports)))
		const fallbackOperations = yield* Queue.takeAll(fallbackReports)
		assert.deepStrictEqual(
			fallbackOperations.map((report) => [report.operation, report.outcome]),
			[
				['open_dm', undefined],
				['post', undefined],
				['post_ephemeral', 'fallback'],
			],
		)
	}),
)
